# ---------------------------------------------------------------------------
# Small helpers
# ---------------------------------------------------------------------------
def now_iso() -> str:
    return datetime.now().isoformat(timespec="seconds")


def new_report_code() -> str:
    return "INC-" + datetime.now().strftime("%y%m%d") + "-" + uuid.uuid4().hex[:5].upper()


def priority_badge(p: str) -> str:
    color = PRIORITY_COLORS.get(p, "#64748b")
    return (
        f'<span class="badge" style="background:{color}1a; color:{color}; '
        f'border:1px solid {color}55;">{p}</span>'
    )


def current_priority(incident_row) -> str:
    return incident_row["human_priority"] or incident_row["suggested_priority"]


# ---------------------------------------------------------------------------
# Database
# ---------------------------------------------------------------------------
def get_conn() -> sqlite3.Connection:
    conn = sqlite3.connect(DB_PATH, check_same_thread=False)
    conn.row_factory = sqlite3.Row
    return conn


def init_db() -> None:
    conn = get_conn()
    c = conn.cursor()
    c.execute(
        """CREATE TABLE IF NOT EXISTS incidents (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            report_code TEXT UNIQUE NOT NULL,
            title TEXT NOT NULL,
            description TEXT NOT NULL,
            location TEXT,

# ---------------------------------------------------------------------------
# Triage engine (deterministic, rule-based)
# ---------------------------------------------------------------------------
def evaluate_triage(d: dict) -> dict:
    """
    Deterministic scoring. Keys read from `d`:
        danger_trapped, danger_rising_water, danger_injury, danger_other (0/1)
        location, description, affected_count, hazard_type
    """
    score = 0
    reasons = []
    flags = []

    if d.get("danger_trapped"):
        score += 50
        reasons.append("Reported people trapped in a dangerous situation.")
    if d.get("danger_rising_water"):
        score += 25
        reasons.append("Reported rapidly rising floodwater.")
    if d.get("danger_injury"):
        score += 30
        reasons.append("Reported serious injury or immediate medical danger.")
    if d.get("danger_other"):
        score += 15
        reasons.append("Other urgent circumstance reported.")

    danger_count = sum(
        1 for k in ("danger_trapped", "danger_rising_water", "danger_injury", "danger_other")
        if d.get(k)
    )
    if danger_count >= 2:
        score += 10
        reasons.append(f"Multiple danger indicators reported ({danger_count}).")

    loc = (d.get("location") or "").strip()
    desc = (d.get("description") or "").lower()
    vague_words = ("somewhere", "unknown", "not sure", "maybe", "possibly", "near")

    if not loc or len(loc) < 6:
        flags.append("Location text is missing or very brief — verify before routing.")
    if loc and any(w in loc.lower() for w in vague_words):
        flags.append("Location contains uncertain language — verify.")
    if desc and any(w in desc for w in vague_words):
        flags.append("Description contains uncertain language — verify.")
    if int(d.get("affected_count", 0) or 0) == 0:
        flags.append("Reported number of affected people is 0 or unspecified — verify scale.")

    if not reasons:
        reasons.append("No immediate-danger indicators reported in the submission.")

    if score >= 60:
        priority = "Urgent Review"
    elif score >= 35:
        priority = "High"
    elif score >= 15:
        priority = "Moderate"
    else:
        priority = "Routine Review"

    if priority == "Urgent Review":
        action = (
            "Human reviewer should verify immediately with the reporter and escalate "
            "to the on-duty coordinator."
        )
    elif priority == "High":
        action = (
            "Reviewer should verify details and prepare a follow-up plan within the "
            "next review cycle."
        )
    elif priority == "Moderate":
        action = "Reviewer should confirm the report and route to the appropriate relief track."
    else:
        action = "Log for routine review; confirm details before routing."

    if d.get("hazard_type") in ("Food", "Water") and danger_count == 0:
        reasons.append(
            "Basic-needs request without immediate-danger indicators — "
            "route to logistics review; do not disregard."
        )

    return {
        "score": score,
        "priority": priority,
        "reasons": reasons,
        "flags": flags,
        "action": action,
    }


# ---------------------------------------------------------------------------
# Duplicate detection
# ---------------------------------------------------------------------------
def _combined_text(title, description, location):
    return " ".join(
        x for x in [title or "", description or "", location or ""] if x
    ).lower()


def similarity_score(text_a: str, text_b: str) -> float:
    seq = SequenceMatcher(None, text_a, text_b).ratio()
    ta = set(re.findall(r"\w+", text_a))
    tb = set(re.findall(r"\w+", text_b))
    overlap = (len(ta & tb) / len(ta | tb)) if (ta and tb) else 0.0
    return 0.5 * seq + 0.5 * overlap


def find_duplicate_pairs(threshold: float = 0.6):
    conn = get_conn()
    incidents = conn.execute(
        "SELECT report_code, title, description, location FROM incidents"
    ).fetchall()
    reviewed = conn.execute(
        "SELECT report_a, report_b FROM duplicate_reviews"
    ).fetchall()
    conn.close()

    reviewed_set = {tuple(sorted((r["report_a"], r["report_b"]))) for r in reviewed}
    pairs = []
    n = len(incidents)
    for i in range(n):
        for j in range(i + 1, n):
            a, b = incidents[i], incidents[j]
            key = tuple(sorted((a["report_code"], b["report_code"])))
            if key in reviewed_set:
                continue
            ta = _combined_text(a["title"], a["description"], a["location"])
            tb = _combined_text(b["title"], b["description"], b["location"])
            score = similarity_score(ta, tb)
            if score >= threshold:
                pairs.append(
                    {
                        "a": a["report_code"],
                        "b": b["report_code"],
                        "a_title": a["title"],
                        "b_title": b["title"],
                        "a_desc": a["description"],
                        "b_desc": b["description"],
                        "location": a["location"] or b["location"],
                        "score": round(score, 3),
                    }
                )
    pairs.sort(key=lambda x: -x["score"])
    return pairs


def record_duplicate_review(a: str, b: str, decision: str) -> None:
    a, b = sorted((a, b))
    conn = get_conn()
    conn.execute(
        "INSERT OR REPLACE INTO duplicate_reviews (report_a, report_b, decision, reviewed_at) "
        "VALUES (?,?,?,?)",
        (a, b, decision, now_iso()),
    )
    conn.commit()
    conn.close()
    log_activity("duplicate_review", f"{a}|{b}", decision,
                 f"Pair reviewed: {a} ↔ {b} — {decision}.")

def compute_match(item_id: str, qty: int):
    conn = get_conn()
    item = conn.execute("SELECT * FROM inventory WHERE item_id=?", (item_id,)).fetchone()
    conn.close()
    if not item:
        return None
    available = int(item["quantity"])
    matchable = min(available, qty)
    shortfall = max(0, qty - available)
    return {
        "available": available,
        "matchable": matchable,
        "shortfall": shortfall,
        "location": item["location"],
        "unit": item["unit"],
        "item_name": item["item_name"],
    }


def fulfill_request(req_id: int):
    conn = get_conn()
    req = conn.execute("SELECT * FROM relief_requests WHERE id=?", (req_id,)).fetchone()
    if not req:
        conn.close()
        return False, "Request not found."
    if req["status"] == "Fulfilled":
        conn.close()
        return False, "Request already fulfilled."
    item = conn.execute(
        "SELECT * FROM inventory WHERE item_id=?", (req["item_id"],)
    ).fetchone()
    if not item:
        conn.close()
        return False, "Inventory item not found."
    deduct = min(int(req["matched_qty"]), int(item["quantity"]))
    if deduct <= 0:
        conn.close()
        return False, "No stock available to deduct."
    new_qty = int(item["quantity"]) - deduct
    try:
        conn.execute(
            "UPDATE inventory SET quantity=?, last_updated=? WHERE item_id=?",
            (new_qty, now_iso(), req["item_id"]),
        )
        conn.execute(
            "UPDATE relief_requests SET status='Fulfilled', reviewed_at=? WHERE id=?",
            (now_iso(), req_id),
        )
        conn.execute(
            "INSERT INTO activity_log (timestamp, entity_type, entity_id, action, details) "
            "VALUES (?,?,?,?,?)",
            (
                now_iso(), "relief_request", str(req_id), "Fulfilled",
                f"Deducted {deduct} {item['unit']} of {item['item_name']} "
                f"from inventory {req['item_id']}. Remaining: {new_qty}.",
            ),
        )
        conn.commit()
        conn.close()
        return True, f"Deducted {deduct} {item['unit']}. Remaining: {new_qty}."
    except sqlite3.IntegrityError as exc:
        conn.rollback()
        conn.close()
        return False, f"Deduction failed: {exc}"

def bar_chart_html(counts: dict, keys: list, colors: dict = None, accent: str = "#2563eb") -> str:
    total = max(sum(counts.get(k, 0) for k in keys), 1)
    html = ""
    for k in keys:
        n = counts.get(k, 0)
        pct = n / total * 100
        color = (colors or {}).get(k, accent)
        html += (
            f"<div style='margin-bottom:10px;'>"
            f"<div style='display:flex; justify-content:space-between; font-size:12px; "
            f"color:#475569; margin-bottom:4px;'><span>{k}</span><span>{n}</span></div>"
            f"<div style='background:#e2e8f0; border-radius:6px; height:10px; overflow:hidden;'>"
            f"<div style='width:{pct:.1f}%; background:{color}; height:100%;'></div>"
            f"</div></div>"
        )
    return html

    ts = now_iso()

    conn = get_conn()
    conn.execute(
        """INSERT INTO incidents (
            report_code, title, description, location, hazard_type, affected_count,
            danger_trapped, danger_rising_water, danger_injury, danger_other,
            reporter, created_at, status, suggested_priority, priority_reasons,
            verify_flags, recommended_action, human_decision, human_priority,
            reviewer_notes, verified_status
        ) VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)""",
        (
            code, title, description, payload["location"], hazard, int(affected),
            payload["danger_trapped"], payload["danger_rising_water"],
            payload["danger_injury"], payload["danger_other"],
            (reporter or "").strip(), ts, "Under Review", tr["priority"],
            json.dumps(tr["reasons"]), json.dumps(tr["flags"]), tr["action"],
            "Pending", tr["priority"], "", "Unverified",
        ),
    )
    conn.commit()
    conn.close()
    log_activity("incident", code, "Report submitted",
                 f"New report: {title} | Hazard: {hazard} | Suggested: {tr['priority']}")

    st.success(
        f"Report submitted successfully. Generated report ID: **{code}** "
        f"· Suggested priority: **{tr['priority']}** (pending human review)."
    )


def browse_reports():
    st.markdown("##### Browse and filter reports")
    conn = get_conn()
    incidents = conn.execute(
        "SELECT * FROM incidents ORDER BY created_at DESC"
    ).fetchall()
    conn.close()

    c1, c2, c3, c4 = st.columns(4)
    with c1:
        f_hazard = st.selectbox("Hazard", ["All"] + HAZARDS)
    with c2:
        f_status = st.selectbox("Status", ["All"] + STATUSES)
    with c3:
        f_priority = st.selectbox("Current priority", ["All"] + PRIORITIES)
    with c4:
        f_search = st.text_input("Search title or location")

    rows = []
    for i in incidents:
        if f_hazard != "All" and i["hazard_type"] != f_hazard:
            continue
        if f_status != "All" and i["status"] != f_status:
            continue
        if f_priority != "All" and current_priority(i) != f_priority:
            continue
        if f_search and f_search.lower() not in (
            (i["title"] + " " + (i["location"] or "")).lower()
        ):
            continue
        rows.append(i)

    st.caption(f"{len(rows)} of {len(incidents)} reports match the current filters.")

    if not rows:
        st.info("No reports match these filters.")
        return

    for i in rows:
        label = (
            f"{i['report_code']} · {i['title']} — "
            f"{current_priority(i)} · {i['status']}"
        )
        with st.expander(label):
            render_incident_detail(i)
            if i["human_decision"] != "Pending":
                st.markdown("---")
                st.markdown("**Update case workflow (administrative only):**")
                with st.form(f"status_form_{i['report_code']}"):
                    new_status = st.selectbox(
                        "Status", STATUSES,
                        index=STATUSES.index(i["status"]),
                        key=f"st2_{i['report_code']}",
                    )
                    note = st.text_input("Note (optional)", key=f"note2_{i['report_code']}")
                    upd = st.form_submit_button("Update Status")
                if upd:
                    conn = get_conn()
                    conn.execute(
                        "UPDATE incidents SET status=? WHERE report_code=?",
                        (new_status, i["report_code"]),
                    )
                    conn.commit()
                    conn.close()
                    log_activity(
                        "incident", i["report_code"], "Status updated",
                        f"New status: {new_status}. Note: {note or 'n/a'}.",
                    )
                    st.success("Status updated.")


def duplicates_tab():
    st.markdown("##### Possible duplicate reports")
    st.caption(
        "Similar wording alone does not prove two reports describe the same event. "
        "Reviewers decide. Confirmed pairs are not shown again as fresh alerts."
    )
    threshold = st.slider(
        "Similarity threshold", min_value=0.4, max_value=0.95, value=0.6, step=0.05
    )
    st.caption(
        f"Score = 0.5 × character-sequence similarity + 0.5 × token overlap. "
        f"Pairs with score ≥ {threshold:.2f} are flagged."
    )

    pairs = find_duplicate_pairs(threshold=threshold)
    if not pairs:
        st.success("No unresolved candidate pairs above the threshold.")
        return

    for idx, p in enumerate(pairs):
        with st.container(border=True):
            st.markdown(
                f"**Combined score: {p['score']:.2f}** · "
                f"`{p['a']}` ↔ `{p['b']}`"
            )
            c1, c2 = st.columns(2)
            with c1:
                st.markdown(f"**{p['a']} — {p['a_title']}**")
                st.write(p["a_desc"])
            with c2:
                st.markdown(f"**{p['b']} — {p['b_title']}**")
                st.write(p["b_desc"])
            st.markdown(f"**Location:** {p['location'] or '—'}")
            st.markdown(
                f"**Reason flagged:** combined similarity {p['score']:.2f} "
                f"≥ threshold {threshold:.2f}. Reviewer verification required."
            )
            b1, b2 = st.columns(2)
            with b1:
                if st.button("Mark as Duplicate", key=f"dup_yes_{idx}"):
                    record_duplicate_review(p["a"], p["b"], "Reviewed — Duplicate")
                    st.success("Pair marked as reviewed duplicate.")
            with b2:
                if st.button("Mark as Separate Reports", key=f"dup_no_{idx}"):
                    record_duplicate_review(p["a"], p["b"], "Reviewed — Separate Reports")
                    st.success("Pair marked as separate reports.")


def page_triage_review():
    st.subheader("Triage Review")
    st.caption(
        "Deterministic rule-based triage. Final decisions remain with trained "
        "human responders. Suggestions may be wrong or incomplete."
    )

    conn = get_conn()
    pending = conn.execute(
        "SELECT * FROM incidents WHERE human_decision='Pending' ORDER BY created_at ASC"
    ).fetchall()
    conn.close()

    if not pending:
        st.success("No reports pending human review.")
        return

    st.markdown(f"**{len(pending)} report(s) awaiting human review.**")
    for i in pending:
        current = current_priority(i)
        with st.expander(f"{i['report_code']} · {i['title']} · {current}"):
            render_incident_detail(i)
            st.markdown("---")
            with st.form(f"review_form_{i['report_code']}"):
                new_priority = st.selectbox(
                    "Final priority decision",
                    PRIORITIES,
                    index=PRIORITIES.index(current),
                    key=f"pr_{i['report_code']}",
                )
                new_status = st.selectbox(
                    "Case status",
                    STATUSES,
                    index=STATUSES.index(i["status"]),
                    key=f"st_{i['report_code']}",
                )
                v_index = (
                    VERIFY_STATUSES.index(i["verified_status"])
                    if i["verified_status"] in VERIFY_STATUSES else 0
                )
                new_verified = st.selectbox(
                    "Field verification",
                    VERIFY_STATUSES,
                    index=v_index,
                    key=f"vf_{i['report_code']}",
                )
                notes = st.text_area(
                    "Reviewer notes (required to record a decision)",
                    key=f"notes_{i['report_code']}",
                )
                submitted = st.form_submit_button("Record Review Decision")

            if submitted:
                if not (notes or "").strip():
                    st.error("Reviewer notes are required to record a decision.")
                else:
                    decision = (
                        "Confirmed"
                        if new_priority == i["suggested_priority"]
                        else "Changed by reviewer"
                    )
                    conn = get_conn()
                    conn.execute(
                        """UPDATE incidents
                           SET human_decision=?, human_priority=?, status=?,
                               reviewer_notes=?, verified_status=?
                           WHERE report_code=?""",

                            decision, new_priority, new_status,
                            notes.strip(), new_verified, i["report_code"],
                        ),
                    )
                    conn.commit()
                    conn.close()

                        "incident", i["report_code"], "Human review recorded",
                        f"Decision: {decision}. Final priority: {new_priority}. "
                        f"Status: {new_status}. Verification: {new_verified}. "
                        f"Notes: {notes.strip()}",
                    )
                    st.success("Review decision saved.")

def new_request_tab():
    st.markdown("##### Create a relief request")
    st.caption(
        "A match is a suggestion. Stock is NOT deducted until a reviewer "
        "explicitly marks the request as Fulfilled."
    )

    conn = get_conn()
    incidents = conn.execute(
        "SELECT report_code, title FROM incidents ORDER BY created_at DESC"
    ).fetchall()
    items = conn.execute(
        "SELECT item_id, item_name, unit, quantity FROM inventory ORDER BY item_id"
    ).fetchall()
    conn.close()

    if not items:
        st.warning("Add an inventory item before creating a request.")
        return

    incident_options = ["(none)"] + [f"{x['report_code']} — {x['title']}" for x in incidents]
    item_options = [
        f"{x['item_id']} — {x['item_name']} ({x['quantity']} {x['unit']})"
        for x in items
    ]

    with st.form("new_relief_request"):
        incident_sel = st.selectbox("Related incident (optional)", incident_options)
        item_sel = st.selectbox("Requested item", item_options)
        qty = st.number_input("Requested quantity", min_value=1, step=1, value=1)
        notes = st.text_area("Request notes (optional)", height=80)
        submitted = st.form_submit_button("Create Request & Match")

    if not submitted:
        return

    if qty <= 0:
        st.error("Requested quantity must be positive.")
        return

    item_id = item_sel.split(" — ")[0].strip()
    incident_code = (
        "" if incident_sel == "(none)" else incident_sel.split(" — ")[0].strip()
    )

    match = compute_match(item_id, int(qty))
    if match is None:
        st.error("Item not found in inventory.")
        return

    explanation = (
        f"Requested {qty} {match['unit']} of {match['item_name']}. "
        f"Available at {match['location'] or 'unspecified location'}: "
        f"{match['available']} {match['unit']}. "
        f"Potentially matchable: {match['matchable']} {match['unit']}. "
        f"Shortfall: {match['shortfall']} {match['unit']}. "
        f"Stock is NOT deducted until a reviewer marks this request as Fulfilled."
    )

    conn = get_conn()
    cur = conn.execute(
        """INSERT INTO relief_requests
           (incident_code, item_id, quantity_requested, notes, status,
            matched_qty, shortfall, storage_location, explanation,
            created_at, reviewed_at)
           VALUES (?,?,?,?,?,?,?,?,?,?,?)""",
        (
            incident_code, item_id, int(qty), notes.strip(), "Pending Review",
            match["matchable"], match["shortfall"], match["location"],
            explanation, now_iso(), None,
        ),
    )
    req_id = cur.lastrowid
    conn.commit()
    conn.close()
    log_activity(
        "relief_request", str(req_id), "Request created",
        f"Requested {qty} {match['unit']} of {item_id}. Matchable: "
        f"{match['matchable']}. Shortfall: {match['shortfall']}.",
    )

    st.success(f"Request #{req_id} created.")
    st.markdown("**Match calculation:**")
    st.markdown(
        f"- Requested: **{qty} {match['unit']}**\n"
        f"- Available: **{match['available']} {match['unit']}**\n"
        f"- Potentially matchable: **{match['matchable']} {match['unit']}**\n"
        f"- Shortfall: **{match['shortfall']} {match['unit']}**\n"
        f"- Suggested storage location: **{match['location'] or '—'}**\n"
        f"- {explanation}"
    )
